    3rd:
        We know that when we need to invoke a function on anything change in React we use the 'useEffect' hook. since the fetch is also like a function, so we need to invoke the fetch api inside the 'useEffect' hook, and set the currency for dependency, and invoke the fetch api in the callback function and convert the response to json like this:
        function useCurrencyInfo(currency){
            useEffect(() => {
                fetch(https://cdn.jsdelivr.net/npm/@fawazahmed0/currency-api@latest/v1/currencies/${currency}.json)
                    .then(res => res.json())
            }, [currency])
        }

        4th:
            Now we need to store the data, so if we store it in a normal variable it will be not change in the UI, that's why we will use the useState hook to store the data and return it like this:
            function useCurrencyInfo(currency){
                const [data, setData] = useState({});
            useEffect(() => {
                fetch(https://cdn.jsdelivr.net/npm/@fawazahmed0/currency-api@latest/v1/currencies/${currency}.json)
                        .then(res => res.json())
                        .then(res => setData(res[currency]))
                }, [currency])
                return data;
            }   

            !important: Don't forget to import the hooks from core react like this:
            import {useEffect, useState} from 'react';

2. Here is it:
    import { useState, useId } from "react";


function InputBox({
    label,
    amount,
    onAmountChange,
    onTypeChange,
    currencyOptions = [],
    selectCurrency = "bdt",
    amountDisable = false,
    currencyDisable = false,
    className = ''
}) {
  const amountInputId = useId()
  return (
    <div className={`bg-white p-3 rounded-lg text-sm flex ${className}`}>
      <div className="w-1/2">
        <label htmlFor={amountInputId} className="text-black/40 mb-2 inline-block">${label}</label>
        <input
          id={amountInputId}
          className="outline-none w-full bg-transparent py-1.5"
          type="number"
          placeholder="Amount"
          value={amount}
          disabled={amountDisable}
          onChange={(e) => {
            return onAmountChange && onAmountChange(Number(e.target.value));
          }}
        />
      </div>
      <div className="w-1/2 flex flex-wrap justify-end text-right">
        <p className="text-black/40 mb-2 w-full">Currency Type</p>
        <select
          className="rounded-lg px-1 py-1 bg-gray-100 cursor-pointer outline-none"
          value={selectCurrency}
          onChange={(e) => onTypeChange && onTypeChange(e.target.value)}
          disabled={currencyDisable}
        >
                  {currencyOptions.map(currency => (
                      <option key={currency} value={currency}>{ currency }</option>
              
          ))}

        </select>
      </div>
    </div>
  );
}

export default InputBox;
